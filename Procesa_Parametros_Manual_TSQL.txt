-- ============================================================================
-- Se ejecutan los bloques por separado, en la conexión indicada en su título.
-- Reemplazar manualmente los valores de las variables tras cada SELECT.
-- ============================================================================

-- PASO 1. CONEXIÓN Tomser
-- Buscar solicitudes confirmadas. Repetir PASOS 2 a 7 POR CADA FOLIO devuelto.
SELECT id_solicitud
FROM [tomser].[dbo].[solicitudes_parametros]
WHERE status = 'CONFIRMADO';

-- PASO 2. CONEXIÓN InterfacesT24
-- Tomar UN folio del PASO 1 y trabajar manualmente los formatos
-- @IndiceFormato = 1, 2, 3, ... hasta 23, EN ESE ORDEN.
-- Para cada índice, ejecutar el SELECT siguiente y luego los PASOS 3 a 6.
DECLARE @FolioConsulta BIGINT = NULL;  -- PEGAR el id_solicitud del PASO 1.
DECLARE @IndiceFormato INT = NULL;    -- PEGAR 1, luego 2, ... y así hasta 23.

SELECT
    B.Descripcion,       -- valor 1
    B.Id_Solicitud,      -- valor 2
    B.Id_Formato,        -- valor 3
    B.Formato,           -- valor 4
    A.Id_Cliente,       -- valor 5
    B.TipoFormato,       -- valor 6
    B.MontoOperacion,    -- valor 7
    B.Moneda,            -- valor 8
    B.Firma              -- valor 9
FROM [tomser].[dbo].[SOLICITUDES_PARAMETROS] AS A
INNER JOIN [tomser].[dbo].[SOLICITUDES_PARAMETROS_DETALLE] AS B
    ON A.ID_SOLICITUD = B.Id_Solicitud
WHERE A.ID_SOLICITUD = @FolioConsulta
  AND B.ID_FORMATO = @IndiceFormato
ORDER BY B.Id_Formato;

-- Guardar los nueve valores devueltos en el orden mostrado.

-- PASO 3. CONEXIÓN Bot
-- Eliminar parametros_cliente usando los valores 3 y 5.
-- Reemplazar los NULL con los valores vigentes del PASO 2.
DECLARE @IdFormato_Bot_Delete BIGINT = NULL;  -- valor 3: B.Id_Formato
DECLARE @IdCliente_Bot_Delete BIGINT = NULL;  -- valor 5: A.Id_Cliente

DELETE FROM [bot].[dbo].[parametros_cliente]
WHERE id_formato = @IdFormato_Bot_Delete
  AND id_cliente = @IdCliente_Bot_Delete;

-- PASO 4. CONEXIÓN tomser
-- Eliminar en tomser, con los mismos valores 3 y 5.
DECLARE @IdFormato_Tomser_Delete BIGINT = NULL; -- valor 3
DECLARE @IdCliente_Tomser_Delete BIGINT = NULL; -- valor 5

DELETE FROM [tomser].[dbo].[parametros_cliente]
WHERE id_formato = @IdFormato_Tomser_Delete
  AND id_cliente = @IdCliente_Tomser_Delete;

-- PASO 5. CONEXIÓN Bot
-- Insertar los nueve valores en el orden EXACTO del original.
-- Copiar de PASO 2 o conservar los valores anteriores si no hubo registros.

DECLARE @Descripcion_Bot VARCHAR(MAX) = NULL;          -- valor 1
DECLARE @IdSolicitud_Bot BIGINT = NULL;               -- valor 2
DECLARE @IdFormato_Bot BIGINT = NULL;                 -- valor 3
DECLARE @Formato_Bot VARCHAR(MAX) = NULL;            -- valor 4
DECLARE @IdCliente_Bot BIGINT = NULL;                 -- valor 5
DECLARE @TipoFormato_Bot INT = NULL;                  -- valor 6
DECLARE @MontoOperacion_Bot DECIMAL(10,0) = NULL;     -- valor 7
DECLARE @Moneda_Bot CHAR(3) = NULL;                   -- valor 8
DECLARE @Firma_Bot INT = NULL;                        -- valor 9

INSERT INTO [bot].[dbo].[parametros_cliente]
VALUES
(
    @Descripcion_Bot, @IdSolicitud_Bot, @IdFormato_Bot,
    @Formato_Bot, @IdCliente_Bot, @TipoFormato_Bot,
    @MontoOperacion_Bot, @Moneda_Bot, @Firma_Bot
);

-- PASO 6. CONEXIÓN tomser
-- Insertar los MISMOS nueve valores en tomser.
-- Reemplazar con los datos usados en PASO 5; ambas conexiones son independientes.
DECLARE @Descripcion_Tomser VARCHAR(MAX) = NULL;       -- valor 1
DECLARE @IdSolicitud_Tomser BIGINT = NULL;            -- valor 2
DECLARE @IdFormato_Tomser BIGINT = NULL;              -- valor 3
DECLARE @Formato_Tomser VARCHAR(MAX) = NULL;         -- valor 4
DECLARE @IdCliente_Tomser BIGINT = NULL;              -- valor 5
DECLARE @TipoFormato_Tomser INT = NULL;               -- valor 6
DECLARE @MontoOperacion_Tomser DECIMAL(10,0) = NULL;  -- valor 7
DECLARE @Moneda_Tomser CHAR(3) = NULL;                -- valor 8
DECLARE @Firma_Tomser INT = NULL;                     -- valor 9

INSERT INTO [tomser].[dbo].[parametros_cliente]
VALUES
(
    @Descripcion_Tomser, @IdSolicitud_Tomser, @IdFormato_Tomser,
    @Formato_Tomser, @IdCliente_Tomser, @TipoFormato_Tomser,
    @MontoOperacion_Tomser, @Moneda_Tomser, @Firma_Tomser
);

-- Repetir PASOS 2, 3, 4, 5 y 6 para @IndiceFormato = 1..23.
-- Orden estricto por índice: SELECT -> DELETE Bot -> DELETE tomser ->
-- INSERT Bot -> INSERT tomser. 

-- PASO 7. CONEXIÓN InterfacesT24
-- SOLO DESPUÉS del índice 23: actualizar estado de la solicitud.
DECLARE @IdSolicitud_Final BIGINT = NULL; -- valor 2: B.Id_Solicitud

UPDATE [tomser].[dbo].[solicitudes_parametros]
SET status = 'PROCESADO'
WHERE id_solicitud = @IdSolicitud_Final;
