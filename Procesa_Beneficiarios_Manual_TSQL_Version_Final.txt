-- PASO 1 | OBTENER FOLIOS PENDIENTES | Conexión Tomser


SELECT s.id_solicitud, s.formato, s.operacion, s.status, s.origen, s.fecha_aut,
       s.id_solicitud_documento, s.id_formato
FROM [tomser].[dbo].[Solicitudes_Beneficiarios] AS s
WHERE s.status IN ('CONFIRMADO','RECHAZADO')
  AND s.FLG_NOTIFICADO = 1
  AND (ISNULL(s.origen,'') <> 'CMS' OR s.operacion <> 'ALTA'
       OR DATEDIFF(MINUTE,s.fecha_aut,GETDATE()) >
          (SELECT VALOR FROM [InterfacesT24].[dbo].[TB_SRV_PARAMETROS_SISTEMA]
           WHERE ID_PARAMETRO='GralAutorizacion'))
ORDER BY s.id_solicitud;
-- PARA CADA FOLIO DEVUELTO, ESCOGER UNO Y SEGUIR LOS PASOS 2 A 7.
